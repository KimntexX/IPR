cmake_minimum_required(VERSION 3.21)
project(IPRProductionRawBench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT IPR_ENGINE_SOURCE OR NOT IPR_ENGINE_REVISION)
    message(FATAL_ERROR "Explicit immutable engine source and revision are required")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(IPR_DEPLOY_MODE system CACHE STRING "" FORCE)
add_subdirectory("${IPR_ENGINE_SOURCE}" engine EXCLUDE_FROM_ALL)
add_executable(iprProductionRawCapture ../production_capture.cpp)
target_link_libraries(iprProductionRawCapture PRIVATE iprEngine)
target_compile_definitions(iprProductionRawCapture PRIVATE IPR_ENGINE_REVISION="${IPR_ENGINE_REVISION}")
add_executable(cameraDenoiseLeakageProbe ../camera_denoise_leakage.cpp)
target_link_libraries(cameraDenoiseLeakageProbe PRIVATE iprEngine)
target_compile_definitions(cameraDenoiseLeakageProbe PRIVATE IPR_ENGINE_REVISION="${IPR_ENGINE_REVISION}"
    IPR_SOURCE_ROOT_PATH="${IPR_ENGINE_SOURCE}")
add_executable(cameraDenoiseRenderServer "${IPR_ENGINE_SOURCE}/src/tools/render_server.cpp"
    "${IPR_ENGINE_SOURCE}/src/tools/TestRecipes.cpp")
target_include_directories(cameraDenoiseRenderServer PRIVATE "${IPR_ENGINE_SOURCE}/src/tools")
target_link_libraries(cameraDenoiseRenderServer PRIVATE iprEngine)
option(IPR_BENCH_SIGNED_DEVELOPMENT "Enable the F1 signedDevelopment API; off for immutable pre-F1 reference" OFF)
if(IPR_BENCH_SIGNED_DEVELOPMENT)
    target_compile_definitions(iprProductionRawCapture PRIVATE IPR_BENCH_SIGNED_DEVELOPMENT=1)
    add_executable(iprRawFloatContract ../float_contract.cpp)
    target_link_libraries(iprRawFloatContract PRIVATE iprEngine)
    target_compile_definitions(iprRawFloatContract PRIVATE IPR_ENGINE_REVISION="${IPR_ENGINE_REVISION}")
endif()
if(WIN32)
    target_link_libraries(iprProductionRawCapture PRIVATE Psapi)
endif()
