cmake_minimum_required(VERSION 3.21)
project(IPRProductionRawBench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT IPR_ENGINE_SOURCE OR NOT IPR_ENGINE_REVISION)
    message(FATAL_ERROR "Explicit immutable engine source and revision are required")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(IPR_DEPLOY_MODE system CACHE STRING "" FORCE)
add_subdirectory("${IPR_ENGINE_SOURCE}" engine EXCLUDE_FROM_ALL)
add_executable(iprProductionRawCapture ../production_capture.cpp)
target_link_libraries(iprProductionRawCapture PRIVATE iprEngine)
target_compile_definitions(iprProductionRawCapture PRIVATE IPR_ENGINE_REVISION="${IPR_ENGINE_REVISION}")
option(IPR_BENCH_SIGNED_DEVELOPMENT "Enable the F1 signedDevelopment API; off for immutable pre-F1 reference" OFF)
if(IPR_BENCH_SIGNED_DEVELOPMENT)
    target_compile_definitions(iprProductionRawCapture PRIVATE IPR_BENCH_SIGNED_DEVELOPMENT=1)
endif()
if(WIN32)
    target_link_libraries(iprProductionRawCapture PRIVATE Psapi)
endif()
