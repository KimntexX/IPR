cmake_minimum_required(VERSION 3.21)
project(IPRNoiseEstimatorProbe LANGUAGES CXX)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
add_executable(noiseEstimatorProbe ../noise_estimator_probe.cpp ../../src/core/NoiseEstimation.cpp)
target_compile_features(noiseEstimatorProbe PRIVATE cxx_std_17)
set_target_properties(noiseEstimatorProbe PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(noiseEstimatorProbe PRIVATE ${OpenCV_INCLUDE_DIRS} ../../src/core)
target_link_libraries(noiseEstimatorProbe PRIVATE ${OpenCV_LIBS} Qt6::Core)
