cmake_minimum_required(VERSION 3.21)
project(IPRNeuralBench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(IPR_NEURAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${IPR_NEURAL_ROOT}"
    OUTPUT_VARIABLE IPR_NEURAL_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(IPR_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${IPR_NEURAL_ROOT}" engine EXCLUDE_FROM_ALL)
add_executable(iprNeuralCameraCapture capture_camera.cpp)
target_link_libraries(iprNeuralCameraCapture PRIVATE iprEngine)
target_compile_definitions(iprNeuralCameraCapture PRIVATE IPR_NEURAL_REVISION="${IPR_NEURAL_REVISION}")
