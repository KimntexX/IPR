cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(IPRNeuralQualification LANGUAGES CXX)
    include(CTest)
endif()
option(IPR_ENABLE_ONNX_RUNTIME "Enable optional neural RAW denoising" OFF)
add_library(iprNeural STATIC NeuralRawDenoiser.cpp NeuralRawDenoiser.h)
target_compile_features(iprNeural PUBLIC cxx_std_17)
set_target_properties(iprNeural PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(iprNeural PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(IPR_ENABLE_ONNX_RUNTIME)
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/OnnxRuntime.cmake")
    if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        # The application copies the runtime DLLs beside its executable.
        set(IPR_NEURAL_RUNTIME_FILES "${IPR_NEURAL_RUNTIME_FILES}" PARENT_SCOPE)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(neuralRawTests "${CMAKE_CURRENT_LIST_DIR}/../../tests/neural_raw_test.cpp")
    target_link_libraries(neuralRawTests PRIVATE iprNeural)
    add_test(NAME neuralRawContract COMMAND neuralRawTests)
    set_tests_properties(neuralRawContract PROPERTIES TIMEOUT 120)
    if(WIN32 AND IPR_ENABLE_ONNX_RUNTIME)
        foreach(runtime IN LISTS IPR_NEURAL_RUNTIME_FILES)
            add_custom_command(TARGET neuralRawTests POST_BUILD COMMAND ${CMAKE_COMMAND}
                -E copy_if_different "${runtime}" "$<TARGET_FILE_DIR:neuralRawTests>")
        endforeach()
    endif()
    set(IPR_NEURAL_TEST_MODEL "" CACHE FILEPATH "Identity ONNX qualification model")
    set(IPR_NEURAL_SPATIAL_TEST_MODEL "" CACHE FILEPATH "Spatial ONNX qualification model")
    # The real exported network on a real crop (reference written on patch_neural_restoration).
    set(IPR_NEURAL_MODEL "" CACHE FILEPATH "Exported denoising network (ONNX)")
    set(IPR_NEURAL_REFERENCE_CASE "" CACHE PATH "Reference case folder of tests/neural/make_reference_case.py")
    add_executable(neuralModelCaseTests "${CMAKE_CURRENT_LIST_DIR}/../../tests/neural_model_case_test.cpp")
    target_link_libraries(neuralModelCaseTests PRIVATE iprNeural)
    if(IPR_ENABLE_ONNX_RUNTIME AND IPR_NEURAL_MODEL AND IPR_NEURAL_REFERENCE_CASE)
        add_test(NAME neuralModelCase COMMAND neuralModelCaseTests "${IPR_NEURAL_MODEL}" "${IPR_NEURAL_REFERENCE_CASE}")
        set_tests_properties(neuralModelCase PROPERTIES TIMEOUT 300)
    endif()
    if(IPR_ENABLE_ONNX_RUNTIME AND IPR_NEURAL_TEST_MODEL)
        add_test(NAME neuralRawRuntime COMMAND neuralRawTests "${IPR_NEURAL_TEST_MODEL}" "${IPR_NEURAL_SPATIAL_TEST_MODEL}")
        set_tests_properties(neuralRawRuntime PROPERTIES TIMEOUT 120)
    endif()
endif()
